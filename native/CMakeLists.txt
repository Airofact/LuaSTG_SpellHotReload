cmake_minimum_required(VERSION 3.20)
project(dims_tree_sitter_lua LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MINGW)
  message(FATAL_ERROR "Windows x64 MinGW GCC required")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CORE_ROOT "" CACHE PATH "tree-sitter v0.25.10 source root")
set(GRAMMAR_ROOT "" CACHE PATH "tree-sitter-lua v0.4.0 source root")
if(NOT EXISTS "${CORE_ROOT}/lib/src/lib.c" OR NOT EXISTS "${GRAMMAR_ROOT}/src/parser.c")
  message(FATAL_ERROR "Explicit pinned CORE_ROOT and GRAMMAR_ROOT required; see README.md")
endif()
file(SHA256 "${GRAMMAR_ROOT}/src/parser.c" grammar_hash)
if(NOT grammar_hash STREQUAL "2be397c6b5094bb7eddb47f285efec0a65dde3f2cdb485de33dc39084e3c7b80")
  message(FATAL_ERROR "Unexpected grammar parser.c; do not apply CRLF fix to another version")
endif()
file(READ "${GRAMMAR_ROOT}/src/parser.c" grammar)
set(before "    case 6:\n      if (lookahead == 'u') ADVANCE(7);\n")
set(after "    case 6:\n      if (lookahead == '\\r') ADVANCE(49);\n      if (lookahead == 'u') ADVANCE(7);\n")
string(REPLACE "${before}" "${after}" grammar "${grammar}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parser.c" "${grammar}")
file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/parser.c" patched_hash)
if(NOT patched_hash STREQUAL "005d2b0b92a5041a892d11471c20327f48192a82a70271042df0f8113ac387c3")
  message(FATAL_ERROR "CRLF lexer patch did not produce the pinned source")
endif()
add_library(dims_tree_sitter_lua SHARED
  "${CORE_ROOT}/lib/src/lib.c" "${CMAKE_CURRENT_BINARY_DIR}/parser.c"
  "${GRAMMAR_ROOT}/src/scanner.c" src/bridge.c src/exports.def)
target_include_directories(dims_tree_sitter_lua PRIVATE
  "${CORE_ROOT}/lib/include" "${CORE_ROOT}/lib/src" "${GRAMMAR_ROOT}/src")
target_compile_definitions(dims_tree_sitter_lua PRIVATE TREE_SITTER_REUSE_ALLOCATOR)
target_link_options(dims_tree_sitter_lua PRIVATE -static-libgcc)
set_target_properties(dims_tree_sitter_lua PROPERTIES PREFIX "")
